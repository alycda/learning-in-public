fn solution(sequence: Vec<i32>) -> bool {
  let c2 = sequence.windows(2).filter(|v| v[0] >= v[1]).count();
  let c3 = sequence.windows(3).filter(|v| v[0] >= v[2]).count();
  
  c2 <= 1 && c3 <= 1
}
