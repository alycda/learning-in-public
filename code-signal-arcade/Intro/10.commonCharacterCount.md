def solution(s1, s2):
  com = [min(s1.count(i),s2.count(i)) for i in set(s1)]
  return sum(com)

## [[Personal/Computer Science/Rust]]

1. Create a HashMap for each String
2. Loop through (shorter) HashMap, if the other HashMap has a matching key, take the lower number

```rust
use std::collections::HashMap;

fn solution(s1: String, s2: String) -> i32 {
	let mut chars1 = HashMap::new();
	let mut chars2 = HashMap::new();

	s1.chars().for_each(|ch| *chars1.entry(ch).or_insert(0) += 1);
	s2.chars().for_each(|ch| *chars2.entry(ch).or_insert(0) += 1);

	chars1.iter().fold(0,
		|acc, (ch, counter)|
		acc + std::cmp::min(counter, chars2.get(ch).unwrap_or(&0)))
}
```
