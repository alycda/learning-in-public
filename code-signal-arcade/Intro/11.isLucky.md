fn solution(n: i32) -> bool {
  let mut b: Vec<u32> = n.to_string().chars().map(|x| x.to_digit(10).unwrap()).collect();

  let a = b.split_off(b.len() / 2);

  (a.iter().sum::<u32>()) == (b.iter().sum::<u32>())
}
